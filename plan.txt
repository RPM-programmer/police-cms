    Для гаи:
        fines.js (штрафы):
          {
            car — номер машины,
            ouwner — владелец,
            date — дата,
            time — время,
            violation — нарушение,
            punishment — {
                type — как наказали (забрали машину, лишили прав или выдали штраф),
                punishment — само наказание (в зависимости от type разное)
            } — наказание,
            collaborator — сотрудник выдавший штраф (наказание),
            status — выплачен штраф или другое 
          }
