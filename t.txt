2. Базовый код точки входа (index.js)
Чтобы код из вашей презентации работал ровно так, как показано на 5-м слайде, точка входа должна быть асинхронной функцией, которая настраивает защиту, синхронизирует БД и вешает роуты на переданный app: 
javascript

const express = require('express');
const helmet = require('helmet');
const rateLimit = require('express-rate-limit');
const { initDatabase } = require('./models');
const registerRoutes = require('./routes');

/**
 * Инициализация police-cms
 * @param {import('express').Express} app - Экземпляр Express
 * @param {Object} [options] - Настройки БД, префикса роутов и безопасности
 */
async function police_cms(app, options = {}) {
  if (!app || typeof app.use !== 'function') {
    throw new Error('[police-cms] Необходимо передать экземпляр Express: police_cms(app)');
  }

  const prefix = options.prefix || '/api/police-cms';

  // 1. Базовая защита сервера (слайд 3: защита от атак)
  app.use(helmet());
  app.use(express.json({ limit: '1mb' }));

  const limiter = rateLimit({
    windowMs: 15 * 60 * 1000, // 15 минут
    max: options.rateLimit || 200,
    standardHeaders: true,
    legacyHeaders: false,
  });
  app.use(prefix, limiter);

  // 2. Инициализация БД и синхронизация таблиц ведомств
  const db = await initDatabase(options.database);

  // 3. Подключение роутов всех ведомств
  registerRoutes(app, prefix, db, options);

  return db;
}

module.exports = police_cms;

Используйте код с осторожностью.

Теперь логика регистрации полностью понятна. Это полноценный государственный реестр (ЗАГС / Паспортный стол + ЕГРЮЛ/ЕГРИП), где учётные записи делятся на физических лиц и юридических лиц (компании), а само создание аккаунта защищено и интегрировано в общую экосистему ведомств.
Как устроена логика регистрации и модели данных
1. Ветка «Человек» (Физическое лицо)
Регистрация нового гражданина происходит только при участии уполномоченного лица (сотрудника паспортного стола / регистратора), так как именно в этот момент генерируются ключевые идентификаторы системы — UID и PID.

    Входные данные формы / API (POST /api/register/citizen):
        registrarPassword — мастер-пароль или токен сотрудника регистрации (проверяется через bcrypt / crypto.timingSafeEqual).
        firstName, lastName, middleName — Имя, Фамилия, Отчество.
        phone — Номер телефона.
        email — Электронная почта.
        birthDate — Дата рождения.
    Что происходит под капотом:
        Проверка пароля регистратора (и запись в аудит-лог, кто именно зарегистрировал гражданина).
        Генерация уникального UID (User ID — сквозной идентификатор гражданина в системе) и PID (Passport ID — серия/номер паспорта).
        Создание записи в основной таблице Citizens («Пользовательская БД» из вашей презентации).
    Ответ сервера (JSON):
    json

    {
      "success": true,
      "uid": "UID-849201",
      "pid": "MP-4920184",
      "citizen": {
        "firstName": "Иван",
        "lastName": "Иванов",
        "middleName": "Иванович",
        "phone": "+375...",
        "email": "ivan@example.com",
        "birthDate": "1995-05-12"
      }
    }

    Используйте код с осторожностью.
    (При неверном пароле регистратора или дубликате данных возвращается { "success": false, "error": "..." })

2. Ветка «Компания» (Юридическое лицо)
Компанию регистрирует уже существующий в системе гражданин (учредитель), подтверждая свою личность через PID и UID, причём регистрация идёт через заявку (запрос) на одобрение (например, в Инспекционное министерство или Суд/Налоговую).

    Входные данные формы / API (POST /api/register/company):
        ownerUid — UID учредителя (гражданина).
        ownerPid — PID паспорта учредителя.
        companyName — Название компании.
        phone — Контактный телефон компании.
        email — Почта компании.

Ниже приведён полный функциональный разбор всех 7 ведомств из вашего списка и пользовательской БД, спроектированный так, чтобы службы работали как единая взаимосвязанная государственная система через сквозные идентификаторы UID (гражданин), PID (паспорт) и CID (компания).
1. Милиция (Общественная безопасность и правопорядок)
Базовый правоохранительный орган, работающий с гражданами, заявлениями и оперативным розыском.

    Картотека граждан (Досье): просмотр базовой информации по UID/PID, судимостей, текущих статусов и приводов.
    Приём и регистрация заявлений (КУСП): создание обращений от граждан (UID заявителя, UID подозреваемого, описание, статус расследования).
    Протоколы правонарушений: оформление административных протоколов и задержаний с привязкой к UID.
    Оперативный розыск: объявление гражданина в местный/федеральный розыск (автоматически отображается у ГАИ, Охраны и Пограничной службы).
    Передача дел в Суд: формирование обвинительного акта и отправка материалов дела в судебную инстанцию.
    Публикация новостей: сводки происшествий и ориентировки в общую ленту на главной странице.

2. ГАИ (Государственная автомобильная инспекция)
Контроль транспорта, водителей, дорожной безопасности и регистрационных знаков.

    Регистрация транспорта (МРЭО): постановка ТС на учёт, выдача техпаспорта (СТС) и гос. номеров, привязка к UID физлица или CID компании.
    Водительские удостоверения (ВУ): выдача прав, открытие категорий, лишение прав (на срок или бессрочно), проверка статуса ВУ.
    Штрафы и нарушения ПДД: выписка штрафов по госномеру или UID водителя, контроль оплаты (с возможностью интеграции с sequelize-game-bank).
    Розыск и перехват ТС: объявление автомобиля в угон или ориентировку, фиксация на постах.
    Эвакуация и штрафстоянка: оформление задержания ТС, причина, статус выдачи владельцу.
    Техосмотр и спецсигналы: выдача диагностических карт и разрешений на спецсигналы для ведомственного транспорта.

3. Охрана (Департамент охраны / Вневедомственная и физическая охрана)
Защита объектов, сопровождение грузов, лицензирование охранной деятельности и оружия.

    Реестр охраняемых объектов: постановка под охрану зданий, банков, предприятий (CID) и частной собственности (UID), статусы сигнализации («Тревога» / «Под охраной»).
    Пропускной режим и допуски: выдача и аннулирование пропусков на режимные объекты и гос. учреждения по UID/PID.
    Лицензии на оружие и ЧОП: выдача разрешений гражданам (UID) на владение/ношение оружия и лицензий частным охранным предприятиям (CID).
    Журнал выездов ГБР (Групп быстрого реагирования): фиксация срабатываний тревожной кнопки, отчёты по выездам и задержаниям на объектах.
    Сопровождение грузов и инкассации: регистрация маршрутов охраны ценных грузов.

4. Пограничная служба
Контроль государственной границы, миграционных потоков и таможенно-пограничного режима.

    Пограничный контроль (КПП): фиксация фактов въезда и выезда граждан по PID и UID (дата, пункт пропуска, направление, транспорт).
    Стоп-лист (Запрет на выезд/въезд): автоматическая блокировка пересечения границы, если на UID/PID висит розыск от Милиции/КГБ, постановление Суда или неоплаченные крупные долги.
    Визы, ВНЖ и миграционный учёт: выдача разрешений на пребывание иностранцам, регистрация временного проживания.
    Досмотр транспорта и грузов: фиксация перевозимых через границу товаров (связь с CID компаний-перевозчиков) и изъятие контрабанды.
    Режим пограничной зоны: выдача пропусков гражданам для нахождения в приграничной полосе.

5. Суд (Судебная система)
Высшая инстанция разрешения споров, вынесения приговоров и санкционирования действий силовиков.

    Судопроизводство (Уголовные, Административные, Гражданские дела): приём дел от Милиции, КГБ, Инспекции и исков от граждан/компаний.
    Вынесение приговоров и решений: назначение сроков заключения, штрафов, конфискации имущества, блокировки счетов компаний (CID).
    Выдача ордеров и санкций: одобрение или отклонение запросов от Милиции и КГБ на арест, обыск, прослушку или заморозку активов (без ордера Суда эти действия в системе могут быть заблокированы).
    Судимости и реабилитация: внесение записи о судимости в досье UID и снятие/погашение судимости.
    Назначение запретов: наложение запрета на выезд за границу (передаётся в Погранслужбу) или запрета на ведение бизнеса.

6. Инспекционное министерство (Госнадзор, Налоговая и Регистрация бизнеса)
Контроль законности деятельности компаний, проверок, налогов и соблюдения регламентов.

    Реестр юридических лиц (Обработка заявок на регистрацию): рассмотрение запросов на создание компаний (проверка UID/PID учредителя и стартового капитала), одобрение или отказ с выдачей CID.
    Плановые и внеплановые проверки: назначение проверок компаниям (CID) и гос. объектам, составление актов нарушений.
    Предписания и приостановка деятельности: выдача обязательных предписаний на устранение нарушений, временное закрытие или ликвидация компании через Суд.
    Налоговый и финансовый контроль: проверка уставного капитала, отчётности компаний, начисление налоговых штрафов (связь с sequelize-game-bank).
    Сертификация и лицензирование бизнеса: выдача разрешений на определённые виды деятельности.

7. КГБ (Комитет государственной безопасности)
Спецслужба высшего уровня доступа с функциями контрразведки, борьбы с организованной преступностью и контроля гос. аппарата.

    Глобальный доступ (Super-Read): просмотр расширенных досье всех граждан (UID/PID), компаний (CID), протоколов Милиции, баз ГАИ и пересечений границы без ограничений.
    Секретные досье и грифы секретности: ведение закрытых дел (Уровень доступа: Секретно / Совершенно секретно), невидимых для обычных сотрудников других ведомств.
    Спецпроверки и допуски к гостайне: проверка кандидатов при приёме на работу в гос. службы, одобрение/отзыв допуска к секретным разделам.
    Оперативно-розыскные мероприятия (ОРМ): постановка граждан (UID) и компаний (CID) на негласный контроль, отслеживание связей и транзакций.
    Антикоррупционный контроль и аудит логов: просмотр журнала действий (AuditLog) сотрудников всех остальных ведомств (кто, когда и чьё досье открывал или менял).
    Федеральный/Государственный розыск высшего приоритета: объявление в особый розыск с мгновенным оповещением всех силовых структур.

8. Пользовательская БД и Личный кабинет (Гражданин + Компания)
То, что доступно обычному авторизованному пользователю (UID / PID):

    Цифровой паспорт: просмотр своего UID, PID, статуса розыска, судимостей и активных запретов (например, запрет на выезд).
    Мои документы и имущество: список зарегистрированных авто (из ГАИ), ВУ, лицензий на оружие (из Охраны), штрафов с кнопкой оплаты.
    Мои компании: список компаний (CID), где гражданин является учредителем, статус заявок на регистрацию, акты проверок от Инспекции.
    Подача обращений: отправка заявления в Милицию, иска в Суд или заявки на регистрацию компании.

    Пограничный режим и Заставы:
        Выдача пропусков в погранзону, журнал патрулирования границы, фиксация нарушений государственной границы и задержания нарушителей.

5. Суд (Правосудие, Конституционный/Арбитражный/Уголовный контроль)
Разделение на коллегии: Уголовная, Административная, Арбитражная (экономическая) и Апелляционная:

    Канцелярия и распределение дел:
        Приём исков от граждан/компаний, обвинительных заключений от Милиции/КГБ и материалов от Инспекции.
        Автоматическое или ручное назначение судьи, назначение даты и времени судебного заседания с уведомлением сторон по UID.
    Электронный зал суда и Протокол заседания:
        Ведение хода заседания, регистрация участников (прокурор, адвокат, подсудимый, свидетели), приобщение ходатайств.
    Система санкций и ордеров (Судебный контроль):
        Выдача цифровых подписанных ордеров: на арест, на обыск жилища/офиса CID, на прослушку/биллинг, на заморозку банковских счетов (sequelize-game-bank).
    Приговоры, Решения и Исполнительные листы:
        Вынесение приговоров (штраф, исправительные работы, лишение свободы, лишение прав/лицензий, конфискация).
        Автоматическая генерация исполнительного листа, который уходит в соответствующие ведомства (в ГАИ — на лишение ВУ, в Погранслужбу — на закрытие выезда, в Инспекцию — на ликвидацию компании).
    Апелляции, Амнистия и Реабилитация:
        Пересмотр дел вышестоящей инстанцией, снятие/погашение судимостей, отмена незаконных штрафов и постановлений других ведомств.

6. Инспекционное министерство (Госнадзор, Налоговая, Регистрация бизнеса и Аудит)

    Единый государственный реестр юрлиц (ЕГРЮЛ):
        Рассмотрение заявок на регистрацию компаний (проверка учредителя UID/PID, проверка легальности стартового капитала, присвоение налогового номера и CID).
        Регистрация изменений в компании (смена директора, изменение уставного капитала, реорганизация, добровольная или принудительная ликвидация).
    Лицензирование видов деятельности и Сертификация:
        Выдача государственных лицензий бизнесу (торговля, медицина, транспорт, строительство, финансы) с контролем срока действия.
    Налоговый контроль и Финансовый мониторинг:
        Учёт налоговых отчислений компаний (CID), выявление уклонения от налогов, блокировка счетов за неуплату (интеграция с sequelize-game-bank).
    Государственные инспекции (Трудовая, Санитарная, Пожарная, Техническая):
        Назначение плановых и внеплановых выездных проверок на объекты компаний (CID) и гос. учреждений.
        Составление чек-листов нарушений, выдача предписаний с дедлайном исправления, опечатывание объектов и приостановка деятельности.
    Госзакупки и Тендеры:
        Публикация государственных заказов от ведомств, подача заявок от зарегистрированных компаний (CID), выбор победителя и контроль исполнения контракта.

7. КГБ (Комитет государственной безопасности — Спецслужба высшего уровня)

    Многоуровневая система грифов секретности:
        Документы, дела и досье с уровнями: ДСП (Для служебного пользования), Секретно, Совершенно секретно, Особой важности.
        Система допусков: даже внутри КГБ сотрудник с 3-й формой допуска не увидит файлы 1-й формы.
    Контрразведка и Оперативно-розыскные мероприятия (ОРМ):
        Постановка объектов (UID граждан, CID компаний, гос. служащих) в «Оперативную разработку».
        Инструменты ОРМ (по санкции или в экстренном порядке): перехват переписок/логов, трекинг пересечений границы, мониторинг банковских переводов (sequelize-game-bank), скрытое наблюдение.
    Спецпроверки и Люстрация гос. аппарата:
        Обязательная проверка кандидатов при приёме на руководящие должности в Милицию, ГАИ, Суд, Охрану, Погранслужбу и Инспекцию.
        Право отстранить любого госслужащего от должности или заблокировать его доступ к базам данных при подозрении в коррупции/госизмене.
    Управление «К» (Кибербезопасность и защита сервера):
        Мониторинг попыток взлома API, брутфорса, несанкционированных запросов к БД, блокировка IP/токенов, просмотр полного AuditLog всех 7 ведомств (кто, кого, когда и зачем пробивал по базе).
    Антитеррор и Чрезвычайное положение (Режим КТО):
        Возможность ввести в системе режим КТО (Контртеррористическая операция) или ЧП: автоматически меняются права доступа на сервере, закрываются границы (Погранслужба), вводятся блокпосты (ГАИ/Милиция) и публикуется экстренное оповещение на главной странице.

8. Сквозные системные механизмы (То, что делает модуль «живым» и сложным)

    Система межведомственного документооборота (СЭД):
        Официальные цифровые документы с уникальным номером, электронной подписью сотрудника, штампом ведомства и QR/хэш-проверкой подлинности (ордера, повестки, справки об отсутствии судимости, лицензии).
    Динамический конструктор ролей и званий (RBAC + ABAC):
        В каждом ведомстве не просто «админ/юзер», а гибкая сетка должностей и прав (например: может читать досье, но не может удалять; может выписать штраф, но не может снять розыск).
    Интеграционные хуки (Event Bus / Webhooks):
        Встроенная шина событий police_cms.events (например, on('citizen:wanted'), on('company:registered'), on('border:crossed'), on('kgb:alert')), чтобы к модулю можно было легко цеплять ботов, банки (sequelize-game-bank), больницы (med-cms) или игровые серверы.
    Продвинутая безопасность сервера (из 3-го слайда вашей презентации):
        Защита от SQL/NoSQL-инъекций и XSS, Rate Limiting на чувствительные ручки, CSRF-токены для веб-интерфейса, шифрование секретных полей КГБ в самой БД (AES-256), защита от перебора паролей регистратора.

Ключевые связи между таблицами (как всё сцеплено в единый организм)

    Citizen (uid, pid) — центральный узел всей БД:

Бэкенд police-cms строится как автономное модульное ядро на CommonJS, которое монтируется внутрь любого Express-приложения одной строкой (await police_cms(app, config)).
Внутри он устроен по слоистой архитектуре (Layered Architecture) с событийно-ориентированным ядром, чтобы 42 таблицы и 7 ведомств работали слаженно, безопасно и не превратились в запутанный код.

2. Как проходит каждый запрос (Жизненный цикл)
Когда пользователь или внешняя система отправляет запрос, бэкенд пропускает его через 6 уровней защиты и обработки:

    Уровень 1: shield.js (Защита периметра)
        Отсекает DDoS/флуд (rate-limit), проверяет заголовки безопасности (helmet), санитизирует входные данные.
        Проверяет SystemState: если КГБ ввело режим КТО или ЧП, автоматически блокируются некритичные гражданские операции (например, выезд за границу или регистрация новых компаний).
    Уровень 2: auth.js (Идентификация)
        Проверяет токен. Подгружает из БД Citizen (кто этот человек по UID) и проверяет таблицу Officer (является ли он действующим сотрудником какого-либо ведомства и не отстранён ли он КГБ).
    Уровень 3: rbac.js (Матрица доступа и секретности)
        Проверяет 3 параметра: Ведомство (например, только gai), Право действия из звания (DepartmentRank.permissions) и Уровень допуска (clearanceLevel 1–3 для секретных документов).
    Уровень 4: Controller -> Service (Бизнес-логика)
        Контроллер проверяет формат полей. Сервис выполняет транзакцию в Sequelize (например, при вынесении приговора Судом обновляет Citizen.criminalRecordsCount и создаёт запись в CourtVerdict).
    Уровень 5: InteragencyEngine + EventBus (Цепная реакция)
        Сервис кидает событие в шину (например, bus.emit('wanted:created', payload)).
        Ядро автоматически срабатывает: добавляет человека в BorderBan (запрет выезда на границе) и включает маячок перехвата на его автомобилях в базе Vehicle (ГАИ).
        Это же событие доступно разработчику снаружи: cms.events.on('wanted:created', ...) для интеграции с ботами или банком (sequelize-game-bank).
    Уровень 6: audit.js (Неизменяемый след)
        Если запрос изменял данные или открывал секретное досье, в AuditLog записывается слепок beforeData и afterData, UID сотрудника и IP-адрес. Удалить или изменить эту запись через API невозможно.

3. Как выглядит подключение и работа с бэкендом в коде пользователя
Разработчик, который поставит ваш пакет через npm i police-cms, сможет использовать его максимально просто, но при желании тонко настраивать:
javascript

const express = require('express');
const police_cms = require('police-cms');

const app = express();

(async () => {
  // Запуск всей системы (БД, 42 таблицы, API, защита, веб-интерфейс)
  const cms = await police_cms(app, {
    database: {
      dialect: 'sqlite', // или 'postgres', 'mysql'
      storage: './gov_database.sqlite',
    },
    security: {
      registrarPassword: process.env.REGISTRAR_PASS, // пароль для регистрации новых граждан
      jwtSecret: process.env.JWT_SECRET,
      encryptionKey: process.env.KGB_AES_KEY,        // 32-байтный ключ шифрования архивов КГБ
    },
    ui: true, // включить встроенный веб-интерфейс по вашему макету
  });

  // Пример подписки на события гос. системы (например, для связки с sequelize-game-bank)
  cms.events.on('company:approved', async ({ company }) => {
    console.log(`Компания ${company.name} (${company.cid}) одобрена Инспекцией!`);
  });

  app.listen(3000, () => console.log('Гос. система запущена на порту 3000'));
})();

Используйте код с осторожностью.
4. Ключевые технические фишки бэкенда

    Транзакционность (ACID): все сложные операции (одобрение компании, передача дела из Милиции в Суд, арест счетов) обёрнуты в sequelize.transaction(), чтобы при сбое база не осталась в полусломанном состоянии.
    Аппаратное шифрование архивов КГБ: поле encryptedPayload в таблице ClassifiedFile шифруется до записи в БД через AES-256-GCM. Даже если кто-то скачает файл базы данных напрямую с диска, прочитать секретные досье без encryptionKey невозможно.
    Единый формат ответов API: любой эндпоинт всегда возвращает предсказуемую структуру { success: true/false, data: ..., error: ..., meta: { timestamp, requestId } }.

const crypto = require('crypto');

class IdGenerator {
  static randomDigits(length = 6) {
    const min = Math.pow(10, length - 1);
    const max = Math.pow(10, length) - 1;
    return crypto.randomInt(min, max).toString();
  }

  static generateUID() {
    return `UID-${this.randomDigits(6)}`;
  }

  static generatePID() {
    const letters = 'ABCEHKMOPTX';
    const prefix = letters[crypto.randomInt(0, letters.length)] + letters[crypto.randomInt(0, letters.length)];
    return `${prefix}-${this.randomDigits(7)}`;
  }

  static generateCID() {
    return `CID-${this.randomDigits(6)}`;
  }

  static generateDocNumber(deptCode = 'GOV') {
    const year = new Date().getFullYear();
    return `${deptCode.toUpperCase()}-${year}-${this.randomDigits(6)}`;
  }
}

module.exports = IdGenerator;

4. core/SecurityManager.js (Хэширование паролей + шифрование AES-256-GCM для КГБ)
javascript

const crypto = require('crypto');
const bcrypt = require('bcrypt');

class SecurityManager {
  constructor(encryptionKeyHex) {
    if (!encryptionKeyHex || encryptionKeyHex.length !== 64) {
      throw new Error('[police-cms] security.encryptionKey должен быть строкой из 64 hex-символов (32 байта)');
    }
    this.key = Buffer.from(encryptionKeyHex, 'hex');
  }

  async hashPassword(plainPassword) {
    return bcrypt.hash(plainPassword, 10);
  }

  async verifyPassword(plainPassword, hash) {
    return bcrypt.compare(plainPassword, hash);
  }

  verifyRegistrarPassword(inputPassword, expectedPassword) {
    const a = Buffer.from(String(inputPassword || ''));
    const b = Buffer.from(String(expectedPassword || ''));
    if (a.length !== b.length) return false;
    return crypto.timingSafeEqual(a, b);
  }

  // Шифрование секретных архивов КГБ
  encryptSecret(plainText) {
    const iv = crypto.randomBytes(12);
    const cipher = crypto.createCipheriv('aes-256-gcm', this.key, iv);
    const encrypted = Buffer.concat([cipher.update(JSON.stringify(plainText), 'utf8'), cipher.final()]);
    const authTag = cipher.getAuthTag();

    return {
      encryptedPayload: encrypted.toString('hex'),
      iv: iv.toString('hex'),
      authTag: authTag.toString('hex'),
    };
  }

  // Расшифровка архивов КГБ при наличии допуска
  decryptSecret({ encryptedPayload, iv, authTag }) {
    const decipher = crypto.createDecipheriv('aes-256-gcm', this.key, Buffer.from(iv, 'hex'));
    decipher.setAuthTag(Buffer.from(authTag, 'hex'));
    const decrypted = Buffer.concat([
      decipher.update(Buffer.from(encryptedPayload, 'hex')),
      decipher.final(),
    ]);
    return JSON.parse(decrypted.toString('utf8'));
  }
}

module.exports = SecurityManager;

3. Middleware для проверки сессии и защиты страниц (middlewares/auth.js)
Благодаря cookie-сессиям мы можем защищать не только API, но и прямо на сервере блокировать открытие закрытых HTML-страниц (cabinet.html и departments/:dept/panel.html), если человек не вошёл или не работает в этой службе:
javascript

// middlewares/auth.js
function createAuthMiddleware(db, config) {
  // 1. Загрузка текущего пользователя из сессии на каждый запрос
  async function attachUser(req, res, next) {
    req.currentUser = null;
    req.currentOfficer = null;

    if (!req.session || !req.session.uid) {
      return next();
    }

    try {
      const citizen = await db.Citizen.findOne({
        where: { uid: req.session.uid, status: 'active' },
        attributes: { exclude: ['passwordHash'] },
      });

      if (!citizen) {
        req.session.destroy(() => {});
        return next();
      }

      req.currentUser = citizen;

      // Проверяем, является ли гражданин действующим сотрудником ведомства
      const officer = await db.Officer.findOne({
        where: { citizenUid: citizen.uid, isSuspended: false },
        include: [{ model: db.DepartmentRank, as: 'rank' }],
      });

      if (officer) {
        req.currentOfficer = officer;
      }

      next();
    } catch (err) {
      next(err);
    }
  }

  // 2. Защита API: только авторизованный гражданин
  function requireAuthApi(req, res, next) {
    if (!req.currentUser) {
      return res.status(401).json({ success: false, error: 'Требуется авторизация' });
    }
    next();
  }

  // 3. Защита HTML-страниц (редирект на /login, если не вошёл)
  function requireAuthPage(req, res, next) {
    if (!req.currentUser) {
      return res.redirect(`${config.basePath}/login`);
    }
    next();
  }

  // 4. Защита рабочей панели ведомства (panel.html)
  function requireDeptPage(req, res, next) {
    const dept = req.params.dept;
    if (!req.currentUser) {
      return res.redirect(`${config.basePath}/login`);
    }
    // КГБ с формой допуска 1–2 может заходить в панели ведомств для надзора
    const isKgbSupervisor =
      req.currentOfficer &&
      req.currentOfficer.departmentCode === 'kgb' &&
      req.currentOfficer.clearanceLevel <= 2;

    if (!req.currentOfficer || (req.currentOfficer.departmentCode !== dept && !isKgbSupervisor)) {
      return res.status(403).send('Доступ запрещён: вы не являетесь сотрудником данного ведомства');
    }
    next();
  }

  return { attachUser, requireAuthApi, requireAuthPage, requireDeptPage };
}

module.exports = createAuthMiddleware;
